# %% [markdown]
# ## Create a function to compute mean spectra per bin

# %%
def compute_binned_spectra(values, X, bin_edges, min_pix=10):
    n_bins = len(bin_edges)
    n_bands = X.shape[1]

    mean_spectra = np.full((n_bins+1, n_bands), np.nan)
    std_spectra  = np.full((n_bins+1, n_bands), np.nan)
    counts       = np.zeros(n_bins+1, dtype=int)

    # ---- BIN 0: exactly clean (0%) ----
    mask0 = values == 0
    counts[0] = mask0.sum()
    if counts[0] >= min_pix:
        mean_spectra[0] = np.nanmean(X[mask0], axis=0)
        std_spectra[0]  = np.nanstd(X[mask0], axis=0)

    # ---- BINS 1–10: 10% intervals ----
    for i, (low, high) in enumerate(bin_edges, start=1):
        mask = (values > low) & (values <= high + 1e-6)
        counts[i] = mask.sum()
        if counts[i] >= min_pix:
            mean_spectra[i] = np.nanmean(X[mask], axis=0)
            std_spectra[i]  = np.nanstd(X[mask], axis=0)

    return mean_spectra, std_spectra, counts

def plot_zoom(mean_spectra, counts, title, w, wmin, wmax):
    plt.figure(figsize=(10,5))
    mask = (w >= wmin) & (w <= wmax)

    # baseline
    plt.plot(w[mask], mean_clean[mask], linewidth=3,
             label=f"0% (clean, n={X_clean.shape[0]})")

    for b in range(mean_spectra.shape[0]):
        m = mean_spectra[b]
        if np.isnan(m).all():
            continue
        label = f"{bin_labels[b]} (n={counts[b]})"
        plt.plot(w[mask], m[mask], linewidth=2, alpha=0.8, label=label)

    plt.xlabel("Wavelength (nm)")
    plt.ylabel("Reflectance")
    plt.title(title)
    plt.grid(True)
    plt.legend(fontsize=8, ncol=2)
    plt.tight_layout()
    plt.show()

def band_depth(spectrum, w, center, left, right):
    """Simple linear-continuum band depth."""
    # indices nearest to requested wavelengths
    i_c = np.argmin(np.abs(w - center))
    i_l = np.argmin(np.abs(w - left))
    i_r = np.argmin(np.abs(w - right))

    Rc = spectrum[i_c]
    Rl = spectrum[i_l]
    Rr = spectrum[i_r]

    # linear continuum at center
    Rc_cont = Rl + (Rr - Rl) * (w[i_c] - w[i_l]) / (w[i_r] - w[i_l])
    return 1 - Rc / Rc_cont

seep_depth_1700 = []
for b in range(mean_seep.shape[0]):
    m = mean_seep[b]
    if np.isnan(m).all():
        seep_depth_1700.append(np.nan)
    else:
        seep_depth_1700.append(
            band_depth(m, wavelengths, center=1730, left=1660, right=1800)
        )
seep_depth_1700 = np.array(seep_depth_1700)

def compute_binned_spectra(values, X, n_bands, min_pix=5):
    n_bins = len(bin_edges)
    mean_spectra = np.full((n_bins+1, n_bands), np.nan)
    count_per_bin = np.zeros(n_bins+1, dtype=int)

    # Bin 0 = 0% pixels
    mask0 = values == 0
    count_per_bin[0] = mask0.sum()
    if count_per_bin[0] >= min_pix:
        mean_spectra[0,:] = np.nanmean(X[mask0], axis=0)

    # Remaining bins
    for b,(low,high) in enumerate(bin_edges):
        mask = (values > low) & (values <= high+1e-6)
        count = mask.sum()
        count_per_bin[b+1] = count
        if count >= min_pix:
            mean_spectra[b+1,:] = np.nanmean(X[mask], axis=0)

    return mean_spectra, count_per_bin
