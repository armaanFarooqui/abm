# ## Read ENMAP data and Flattening Enmap pixel

# %%
with rasterio.open(enmap_path) as src:
    enmap_raw   = src.read()          # (bands, H, W)
    enmap_nodata = src.nodata
    H, W = src.height, src.width
    n_bands = enmap_raw.shape[0]

print("ENMAP shape (bands, H, W):", enmap_raw.shape, "nodata:", enmap_nodata)

# convert to float and set nodata to NaN
enmap = enmap_raw.astype("float32")
if enmap_nodata is not None:
    enmap[enmap_raw == enmap_nodata] = np.nan


# (bands, H, W) -> (H, W, bands) -> (N_pixels, n_bands)
enmap_hwk = np.moveaxis(enmap, 0, -1)
X_all = enmap_hwk.reshape(-1, enmap_hwk.shape[-1])

print("X_all shape:", X_all.shape)
print("NaNs in X_all before impute:", np.isnan(X_all).sum())

# %%
nan_per_pixel  = np.isnan(X_all).sum(axis=1)
nan_frac_pixel = nan_per_pixel / X_all.shape[1]

good_pixels = nan_frac_pixel < 0.5    # or 0.3, etc.
X_all = X_all[good_pixels, :]


# %%
band_means = np.nanmean(X_all, axis=0)
bad_bands  = np.isnan(band_means)      

X_all      = X_all[:, ~bad_bands]
band_means = band_means[~bad_bands]

nan_rows, nan_cols = np.where(np.isnan(X_all))
X_all[nan_rows, nan_cols] = band_means[nan_cols]
